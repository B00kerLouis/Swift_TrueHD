cmake_minimum_required(VERSION 3.16)
project(TrueHDDecoder VERSION 1.4.2 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(STHD_NATIVE_DEVICE "Discover default output device topology" ON)
option(STHD_BUILD_TESTS "Build C/C++ decoder validation" ON)
option(STHD_BUILD_FUZZER "Build Clang libFuzzer decoder validation" OFF)
if(WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(STHD_LIBRARY_KIND SHARED)
else()
    set(STHD_LIBRARY_KIND STATIC)
endif()
add_library(libtruehdec ${STHD_LIBRARY_KIND}
    Sources/DecoderFramework/Decoder.cpp
    Sources/DecoderFramework/Checksums.cpp
    Sources/DecoderFramework/Renderer.cpp
    Sources/DecoderFramework/PlatformAudio.cpp
    Sources/DecoderFramework/AudioOutput.cpp
    Sources/DecoderFramework/NativeAudio.cpp
    Sources/DecoderFramework/Player.cpp)
target_include_directories(libtruehdec PUBLIC Sources/DecoderFramework/include)
target_compile_features(libtruehdec PUBLIC cxx_std_17)
find_package(Threads REQUIRED)
target_link_libraries(libtruehdec PUBLIC Threads::Threads)
set_target_properties(libtruehdec PROPERTIES POSITION_INDEPENDENT_CODE ON
    OUTPUT_NAME truehdec CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(WIN32)
    set_target_properties(libtruehdec PROPERTIES PREFIX "")
endif()
if(STHD_LIBRARY_KIND STREQUAL "SHARED")
    target_compile_definitions(libtruehdec PUBLIC STHD_SHARED=1 PRIVATE STHD_BUILDING_LIBRARY=1)
    set_target_properties(libtruehdec PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 1)
endif()
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
if(NOT STHD_NATIVE_DEVICE)
    target_compile_definitions(libtruehdec PRIVATE STHD_DISABLE_NATIVE_DEVICE=1)
elseif(APPLE)
    target_link_libraries(libtruehdec PUBLIC "-framework CoreAudio" "-framework AudioToolbox")
elseif(WIN32)
    target_link_libraries(libtruehdec PUBLIC ole32 uuid ksuser)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(PIPEWIRE QUIET IMPORTED_TARGET libpipewire-0.3)
        if(PIPEWIRE_FOUND)
            target_sources(libtruehdec PRIVATE Sources/DecoderFramework/PipeWireAudio.cpp)
            target_compile_definitions(libtruehdec PRIVATE STHD_HAVE_PIPEWIRE=1)
            target_link_libraries(libtruehdec PUBLIC PkgConfig::PIPEWIRE)
        endif()
    endif()
    find_package(ALSA QUIET)
    if(ALSA_FOUND)
        target_compile_definitions(libtruehdec PRIVATE STHD_HAVE_ALSA=1)
        target_include_directories(libtruehdec PRIVATE ${ALSA_INCLUDE_DIRS})
        target_link_libraries(libtruehdec PUBLIC ${ALSA_LIBRARIES})
    endif()
endif()
if(MSVC)
    target_compile_options(libtruehdec PRIVATE /W4 /permissive-)
else()
    target_compile_options(libtruehdec PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(truehdec Sources/DecoderCLI/main.cpp)
target_link_libraries(truehdec PRIVATE libtruehdec)
set_target_properties(truehdec PROPERTIES OUTPUT_NAME truehdec)
if(APPLE)
    target_sources(truehdec PRIVATE Sources/DecoderCLI/macos/QCEngine.cpp
        Sources/DecoderCLI/macos/QCPlayer.cpp)
    target_link_libraries(truehdec PRIVATE "-framework AppKit" "-framework CoreGraphics" objc)
    add_custom_command(TARGET truehdec POST_BUILD
        COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/Sources/DecoderCLI/macos/package.sh"
            "$<TARGET_FILE:truehdec>" "${CMAKE_CURRENT_BINARY_DIR}/truehdec QC.app"
            "${CMAKE_CURRENT_SOURCE_DIR}/Sources/DecoderCLI/macos/Info.plist"
        VERBATIM)
    set_target_properties(truehdec PROPERTIES INSTALL_RPATH "@loader_path/../lib")
elseif(UNIX)
    set_target_properties(truehdec PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
endif()
if(MINGW)
    target_link_options(truehdec PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
if(STHD_BUILD_TESTS)
    enable_testing()
    add_executable(decoder_tests DecoderTests/decoder_tests.cpp DecoderTests/c_api_test.c
        Sources/DecoderFramework/Checksums.cpp DecoderTests/matrix_syntax_tests.cpp)
    target_link_libraries(decoder_tests PRIVATE libtruehdec)
    if(APPLE AND STHD_NATIVE_DEVICE)
        target_compile_definitions(decoder_tests PRIVATE STHD_TEST_COREAUDIO_MAPS=1)
    endif()
    add_test(NAME decoder_tests COMMAND decoder_tests)
    foreach(profile IN ITEMS 71 12 14 16)
        set(fixture "${CMAKE_CURRENT_SOURCE_DIR}/Integrations/FFmpeg/fixtures/synthetic-${profile}.mlp")
        if(EXISTS "${fixture}")
            add_test(NAME decoder_stream_${profile} COMMAND decoder_tests --stream "${fixture}")
        endif()
    endforeach()
    add_test(NAME decoder_matrix COMMAND decoder_tests --matrix "${CMAKE_CURRENT_SOURCE_DIR}/Integrations/FFmpeg/fixtures/synthetic-16.mlp")
    add_test(NAME decoder_motion COMMAND decoder_tests --motion "${CMAKE_CURRENT_SOURCE_DIR}/Integrations/FFmpeg/fixtures/timed-16.mlp")
    add_test(NAME decoder_checksum COMMAND decoder_tests --checksum "${CMAKE_CURRENT_SOURCE_DIR}/Integrations/FFmpeg/fixtures/synthetic-71.mlp")
    add_executable(player_smoke DecoderTests/player_smoke.cpp)
    target_link_libraries(player_smoke PRIVATE libtruehdec)
    add_executable(audio_smoke DecoderTests/audio_smoke.cpp)
    target_link_libraries(audio_smoke PRIVATE libtruehdec)
    if(APPLE)
        add_executable(qc_engine_tests DecoderTests/qc_engine_tests.cpp
            Sources/DecoderCLI/macos/QCEngine.cpp)
        target_link_libraries(qc_engine_tests PRIVATE libtruehdec)
        add_test(NAME decoder_qc_engine COMMAND qc_engine_tests
            "${CMAKE_CURRENT_SOURCE_DIR}/Integrations/FFmpeg/fixtures/synthetic-16.mlp"
            "${CMAKE_CURRENT_BINARY_DIR}/QCValidation")
    endif()
endif()
install(TARGETS libtruehdec truehdec ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
if(STHD_BUILD_FUZZER)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "STHD_BUILD_FUZZER requires Clang with libFuzzer")
    endif()
    add_executable(decoder_fuzz DecoderTests/decoder_fuzz.cpp
        Sources/DecoderFramework/Decoder.cpp Sources/DecoderFramework/Checksums.cpp)
    target_include_directories(decoder_fuzz PRIVATE Sources/DecoderFramework/include)
    target_compile_options(decoder_fuzz PRIVATE -fsanitize=fuzzer,address,undefined
        -fno-omit-frame-pointer)
    target_link_options(decoder_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
install(FILES Sources/DecoderFramework/include/TrueHDDecoder.h DESTINATION include)
install(FILES Decoder/LICENSE LICENSES/LGPL-2.1-or-later.txt
    DESTINATION share/licenses/truehdec)
set(STHD_PC_PRIVATE_LIBS "")
if(NOT MSVC)
    set(STHD_PC_PRIVATE_LIBS "-pthread")
endif()
set(STHD_PC_CFLAGS "")
if(APPLE)
    string(APPEND STHD_PC_PRIVATE_LIBS " -lc++")
    if(STHD_NATIVE_DEVICE)
        string(APPEND STHD_PC_PRIVATE_LIBS " -framework CoreAudio -framework AudioToolbox")
    endif()
elseif(NOT MSVC)
    string(APPEND STHD_PC_PRIVATE_LIBS " -lstdc++ -lm")
    if(WIN32 AND STHD_NATIVE_DEVICE)
        string(APPEND STHD_PC_PRIVATE_LIBS " -lole32 -luuid -lksuser")
    elseif(STHD_NATIVE_DEVICE AND (PIPEWIRE_FOUND OR ALSA_FOUND))
        foreach(native_library IN LISTS PIPEWIRE_LINK_LIBRARIES ALSA_LIBRARIES)
            string(APPEND STHD_PC_PRIVATE_LIBS " ${native_library}")
        endforeach()
    endif()
endif()
if(WIN32)
    set(STHD_PC_CFLAGS " -DSTHD_SHARED=1")
endif()
configure_file(cmake/truehdec.pc.in truehdec.pc @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/truehdec.pc" DESTINATION lib/pkgconfig)
