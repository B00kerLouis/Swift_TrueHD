cmake_minimum_required(VERSION 3.16)
project(TrueHDDecoder VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(STHD_NATIVE_DEVICE "Discover default output device topology" ON)
option(STHD_BUILD_TESTS "Build C/C++ decoder validation" ON)
add_library(decoder_framework STATIC
    Sources/DecoderFramework/Decoder.cpp
    Sources/DecoderFramework/Checksums.cpp
    Sources/DecoderFramework/Renderer.cpp
    Sources/DecoderFramework/PlatformAudio.cpp
    Sources/DecoderFramework/AudioOutput.cpp
    Sources/DecoderFramework/NativeAudio.cpp)
target_include_directories(decoder_framework PUBLIC Sources/DecoderFramework/include)
target_compile_features(decoder_framework PUBLIC cxx_std_17)
find_package(Threads REQUIRED)
target_link_libraries(decoder_framework PUBLIC Threads::Threads)
set_target_properties(decoder_framework PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(NOT STHD_NATIVE_DEVICE)
    target_compile_definitions(decoder_framework PRIVATE STHD_DISABLE_NATIVE_DEVICE=1)
elseif(APPLE)
    target_link_libraries(decoder_framework PUBLIC "-framework CoreAudio" "-framework AudioToolbox")
elseif(WIN32)
    target_link_libraries(decoder_framework PUBLIC ole32 uuid ksuser)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(PIPEWIRE QUIET IMPORTED_TARGET libpipewire-0.3)
        if(PIPEWIRE_FOUND)
            target_sources(decoder_framework PRIVATE Sources/DecoderFramework/PipeWireAudio.cpp)
            target_compile_definitions(decoder_framework PRIVATE STHD_HAVE_PIPEWIRE=1)
            target_link_libraries(decoder_framework PUBLIC PkgConfig::PIPEWIRE)
        endif()
    endif()
    find_package(ALSA QUIET)
    if(ALSA_FOUND)
        target_compile_definitions(decoder_framework PRIVATE STHD_HAVE_ALSA=1)
        target_include_directories(decoder_framework PRIVATE ${ALSA_INCLUDE_DIRS})
        target_link_libraries(decoder_framework PUBLIC ${ALSA_LIBRARIES})
    endif()
endif()
if(MSVC)
    target_compile_options(decoder_framework PRIVATE /W4 /permissive-)
else()
    target_compile_options(decoder_framework PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(decoder_cli Sources/DecoderCLI/main.cpp)
target_link_libraries(decoder_cli PRIVATE decoder_framework)
set_target_properties(decoder_cli PROPERTIES OUTPUT_NAME truehdd)
if(MINGW)
    target_link_options(decoder_cli PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
if(STHD_BUILD_TESTS)
    enable_testing()
    add_executable(decoder_tests DecoderTests/decoder_tests.cpp DecoderTests/c_api_test.c)
    target_link_libraries(decoder_tests PRIVATE decoder_framework)
    add_test(NAME decoder_tests COMMAND decoder_tests)
    add_executable(audio_smoke DecoderTests/audio_smoke.cpp)
    target_link_libraries(audio_smoke PRIVATE decoder_framework)
endif()
install(TARGETS decoder_framework decoder_cli ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(FILES Sources/DecoderFramework/include/TrueHDDecoder.h DESTINATION include)
